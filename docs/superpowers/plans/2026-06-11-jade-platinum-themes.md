**Goal:** Add two new theme families — Jade and Platinum — each as a Dark + Light pair, per the approved spec `docs/superpowers/specs/2026-06-11-jade-platinum-themes-design.md`.

**Architecture:** Theme JSONs are Monaco-format files in `src/shared/themes/data/` consumed by `convertTheme` (`src/shared/themes/adapter.ts`), which derives every UI token from the `colors` block. The four new JSONs are generated by recoloring the Manifold rule structure with per-theme hex maps (the spec's substitution rule, encoded), plus explicit `colors` blocks. Registration is data-driven (`theme-data.ts`); the titlebar family dropdown needs a hardcoded union widened in three renderer files (same as the Royal PR #419).

def recolor(rules, hex_map, overrides):
    out = []
    for rule in rules:
        r = dict(rule)
        for key in ('foreground', 'background'):
            if key in r:
                r[key] = hex_map.get(r[key], r[key])
        token = r.get('token', '')
        if token in overrides:
            r['foreground'] = overrides[token]
        out.append(r)
    return out


def fmt(theme):
    """Match the existing one-rule-per-line file format."""
    lines = ['{', '  "base": "%s",' % theme['base'], '  "inherit": true,', '  "rules": [']
    rule_lines = []
    for r in theme['rules']:
        inner = ', '.join('%s: %s' % (json.dumps(k), json.dumps(v)) for k, v in r.items())
        rule_lines.append('    { ' + inner + ' }')
    lines.append(',\n'.join(rule_lines))
    lines.append('  ],')
    lines.append('  "colors": {')
    lines.append(',\n'.join('    %s: %s' % (json.dumps(k), json.dumps(v)) for k, v in theme['colors'].items()))
    lines.append('  }')
    lines.append('}')
    return '\n'.join(lines) + '\n'

    (DATA / ('%s.json' % out_label)).write_text(fmt(theme))
